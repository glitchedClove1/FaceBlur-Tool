# Trimmed dependency set for the Gradio deployment (Render, Hugging Face
# Spaces free CPU tier) - this is NOT the training environment's
# requirements.txt. Excludes everything training-only: albumentations,
# tensorboard, pytest, gdown, scipy (only used by engine/evaluate.py, not
# the deployed app).
#
# CPU-only torch/torchvision, pinned explicitly with the +cpu local version
# and --extra-index-url. This is not optional: PyPI's default `torch`
# package for Linux bundles the full CUDA runtime (cuBLAS/cuDNN/cuFFT/NCCL/
# triton, several hundred MB) even with no GPU present - confirmed on a
# real deploy, where plain `torch==2.13.0` resolved to the +cu126 build via
# pip and blew through Render's 512MB free-tier limit before the app
# process even finished starting. The +cpu tag forces the unambiguous CPU
# wheel regardless of which resolver runs the install.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.13.0+cpu
torchvision==0.28.0+cpu

numpy==2.5.2
opencv-python-headless==5.0.0.93
pyyaml==6.0.3
gradio==6.26.0
